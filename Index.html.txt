        data: [25, 35, 10, 10, 20], //[cite: 2]

      },
      q3: {
        labels: ['Sites de notícias', 'TikTok', 'Instagram', 'Outros', 'WhatsApp', 'YouTube'],
        data: [35, 25, 20, 10, 5, 5], //[cite: 3]
        colors: ['#10b981', '#ef4444', '#2563eb', '#06b6d4', '#a855f7', '#f59e0b']

      },
      q5: {
        labels: ['Na maioria das vezes', 'Sim, com facilidade', 'Não consigo identificar', 'Tenho dificuldade'],
        data: [50, 25, 20, 5], //[cite: 5]
        colors: ['#ef4444', '#2563eb', '#10b981', '#f59e0b']
      }

    let chartInstances = {};

    function renderCharts(textColor) {
      const commonOptions = {
        responsive: true,
        maintainAspectRatio: false,
        plugins: {
          legend: {
            position: 'bottom',
            labels: { color: textColor, font: { family: 'sans-serif', size: 12 } }
          }
        }
      };

      // Gráfico 1 - Rosca
      if (chartInstances.c1) chartInstances.c1.destroy();
      chartInstances.c1 = new Chart(document.getElementById('chart1'), {
        type: 'doughnut',
        data: {
          labels: surveyData.q1.labels,
          datasets: [{ data: surveyData.q1.data, backgroundColor: surveyData.q1.colors }]
        },
        options: commonOptions
      });

      // Gráfico 2 - Rosca
      if (chartInstances.c2) chartInstances.c2.destroy();
      chartInstances.c2 = new Chart(document.getElementById('chart2'), {
        type: 'doughnut',
        data: {
          labels: surveyData.q2.labels,
          datasets: [{ data: surveyData.q2.data, backgroundColor: surveyData.q2.colors }]
        },
        options: commonOptions
      });

      // Gráfico 3 - Barra
      if (chartInstances.c3) chartInstances.c3.destroy();
      chartInstances.c3 = new Chart(document.getElementById('chart3'), {
        type: 'bar',
        data: {
          labels: surveyData.q3.labels,
          datasets: [{
            label: 'Porcentagem (%)',
            data: surveyData.q3.data,
            backgroundColor: surveyData.q3.colors
          }]
        },
        options: {
          ...commonOptions,
          scales: {
            x: { ticks: { color: textColor }, grid: { display: false } },
            y: { ticks: { color: textColor }, grid: { color: 'rgba(150, 150, 150, 0.1)' } }
          }
        }
      });

      // Gráfico 4 - Rosca
      if (chartInstances.c4) chartInstances.c4.destroy();
      chartInstances.c4 = new Chart(document.getElementById('chart4'), {
        type: 'doughnut',
        data: {
          labels: surveyData.q4.labels,
          datasets: [{ data: surveyData.q4.data, backgroundColor: surveyData.q4.colors }]
        },
        options: commonOptions
      });

      // Gráfico 5 - Barra Horizontal
      if (chartInstances.c5) chartInstances.c5.destroy();
      chartInstances.c5 = new Chart(document.getElementById('chart5'), {
        type: 'bar',
        data: {
          labels: surveyData.q5.labels,
          datasets: [{
            label: 'Porcentagem (%)',
            data: surveyData.q5.data,
            backgroundColor: surveyData.q5.colors
          }]
        },
        options: {
          indexAxis: 'y',
          ...commonOptions,
          scales: {
            x: { ticks: { color: textColor }, grid: { color: 'rgba(150, 150, 150, 0.1)' } },
            y: { ticks: { color: textColor }, grid: { display: false } }
          }
        }
      });
    }

    // Lógica de Alternância de Tema
    const themeToggleBtn = document.getElementById('theme-toggle');
    const themeIcon = document.getElementById('theme-icon');
    const themeText = document.getElementById('theme-text');

    function getTextColor() {
      return document.documentElement.getAttribute('data-theme') === 'dark' ? '#f8fafc' : '#0f172a';
    }

    themeToggleBtn.addEventListener('click', () => {
      const currentTheme = document.documentElement.getAttribute('data-theme');
      const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
      
      document.documentElement.setAttribute('data-theme', newTheme);
      themeIcon.textContent = newTheme === 'dark' ? '☀️' : '🌙';
      themeText.textContent = newTheme === 'dark' ? 'Modo Claro' : 'Modo Escuro';

      renderCharts(getTextColor());
    });

    // Renderização inicial
    renderCharts(getTextColor());
  </script>
